# test_circular.py
import sys
sys.path.insert(0, '.')

def test_circular_imports():
    """Test all model imports together"""
    modules_to_test = [
        'app.models.general',
        'app.models.driver',
        'app.models.delivery',
        'app.models.order',
        'app.models.organization',
        'app.models.restaurant',
    ]

    for module in modules_to_test:
        try:
            __import__(module)
            print(f"✓ {module} imports cleanly")
        except ImportError as e:
            print(f"✗ {module} import error: {e}")
        except Exception as e:
            print(f"✗ {module} circular import? {type(e).__name__}: {e}")

if __name__ == "__main__":
    test_circular_imports()
